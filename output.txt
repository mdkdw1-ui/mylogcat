===== ./app/src/main/java/com/example/mylogcat/LogViewModel.kt =====
package com.example.mylogcat

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.BufferedReader
import java.io.InputStreamReader

class LogViewModel : ViewModel() {
    private val _isVpnOn = MutableStateFlow(false)
    val isVpnOn: StateFlow<Boolean> = _isVpnOn

    private val _hasConf = MutableStateFlow(false)
    val hasConf: StateFlow<Boolean> = _hasConf

    private val _logs = MutableStateFlow<List<String>>(emptyList())
    val logs: StateFlow<List<String>> = _logs

    fun setVpnState(isOn: Boolean) { _isVpnOn.value = isOn }
    fun setHasConf(has: Boolean) { _hasConf.value = has }

    fun addLog(line: String) {
        val current = _logs.value.toMutableList()
        current.add(line)
        if (current.size > 3000) { current.removeAt(0) }
        _logs.value = current
    }

    fun clearLogs() { _logs.value = emptyList() }

    // ★ 앱 실행 시 과거의 'Crash(크래시)' 버퍼 로그를 가져와 화면에 띄움
    suspend fun loadPastCrashLogs() = withContext(Dispatchers.IO) {
        try {
            val process = Runtime.getRuntime().exec(arrayOf("logcat", "-d", "-b", "crash", "-v", "time", "-t", "100"))
            val reader = BufferedReader(InputStreamReader(process.inputStream))
            var line: String?
            val pastLogs = mutableListOf<String>()
            while (reader.readLine().also { line = it } != null) {
                line?.let { pastLogs.add(it) }
            }
            if (pastLogs.isNotEmpty()) {
                _logs.value = pastLogs
            }
        } catch (e: Exception) {}
    }
}

        if (turnOn) {
            val file = File(filesDir, "wg.conf")
            if (!file.exists()) { viewModel.setVpnState(false); return }
            val intent = VpnService.prepare(this)
            if (intent != null) startActivityForResult(intent, 100) else startVpnService(file)

            stopService(Intent(this, WireGuardService::class.java).setAction("stop"))
        }

        startService(Intent(this, WireGuardService::class.java).apply { action = "start"; putExtra("confPath", confFile.absolutePath) })
    }

package com.example.mylogcat

import android.content.Intent
import android.net.VpnService
import android.os.IBinder
import com.wireguard.android.backend.Backend
import com.wireguard.android.backend.GoBackend
import com.wireguard.android.backend.Tunnel
import com.wireguard.config.Config
import java.io.File

class WireGuardService : android.app.Service() {
    private var backend: Backend? = null
    private var tunnel: Tunnel? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            "start" -> {
                val confPath = intent.getStringExtra("confPath") ?: return START_NOT_STICKY
                if (VpnService.prepare(this) != null) { stopSelf(); return START_NOT_STICKY }
                startWg(confPath)
            }
            "stop" -> stopWg()
        }
        return START_NOT_STICKY
    }

    private fun startWg(confPath: String) {
        try {
            val cfg = File(confPath).inputStream().use { Config.parse(it) }
            backend = GoBackend(this)
            tunnel = object : Tunnel {
                override fun getName() = "mylogcat-wg"
                override fun onStateChange(newState: Tunnel.State) {
                    running = (newState == Tunnel.State.UP)
                }
            }
            backend?.setState(tunnel, Tunnel.State.UP, cfg)
            running = true
        } catch (e: Exception) {
            running = false
            stopSelf()
        }
    }

    private fun stopWg() {
        try { if (tunnel != null) backend?.setState(tunnel, Tunnel.State.DOWN, null) } catch (_: Exception) {}
        running = false
        stopSelf()
    }

    override fun onDestroy() { stopWg(); super.onDestroy() }

    companion object { @Volatile var running = false }
}
